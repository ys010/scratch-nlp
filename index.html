בַּלַּשׁ הַמִּילִּים: אֵיךְ מַחְשֵׁב לוֹמֵד לִכְתּוֹב מִשְׁפָּטִים?

דַּמְיְינוּ מַחְשֵׁב שֶׁקּוֹרֵא סֵפֶר שָׁלֵם — וְאָז כּוֹתֵב מִשְׁפָּטִים חֲדָשִׁים מִשֶּׁלּוֹ, בְּלִי שֶׁאַף אֶחָד לִימֵּד אוֹתוֹ עִבְרִית. מוּזָר, נָכוֹן? יֵשׁ כָּאן סוֹד קָטָן, וַאֲנַחְנוּ הוֹלְכִים לְגַלּוֹת אוֹתוֹ כְּמוֹ בַּלָּשִׁים. וְאָז — הֲכִי כֵּיף — נִבְנֶה אוֹתוֹ בְּעַצְמֵנוּ, צַעַד אַחַר צַעַד, בִּסְקָרֵאץ'.

02 · תִּיק הַחֲקִירָה

שְׁמוֹנָה מְשִׂימוֹת עַד שֶׁנִּפְצַח אֶת הַסּוֹד

כָּל מְשִׂימָה מְלַמֶּדֶת טְרִיק קָטָן אֶחָד בְּסָקֵרָאץ', וּמְקָרֶבֶת אוֹתָנוּ צַעַד אֶחָד לַהֲבָנָה אֵיךְ מִילִּים "מִתְחַבְּרוֹת" זוֹ לְזוֹ. מְשִׂימָה 1 פְּתוּחָה עַכְשָׁיו לְמַטָּה; הַשְּׁאָר מְחַכּוֹת בַּהֶמְשֵׁךְ.

1

כַּמָּה מִילִּים יֵשׁ בַּמִּשְׁפָּט, בְּלִי לִסְפּוֹר עַל הָאֶצְבָּעוֹת?

פָּתוּחַ עַכְשָׁיו ↓

הַתּוֹכְנָה הָרִאשׁוֹנָה שֶׁלָּנוּ: מַקְלִידִים מִשְׁפָּט, וְהַמַּחְשֵׁב סוֹפֵר כַּמָּה מִילִּים יֵשׁ בּוֹ — אוֹת אַחֲרֵי אוֹת, כְּמוֹ בַּלָּשׁ שֶׁבּוֹדֵק כָּל פְּרָט בִּזְכוּכִית מַגְדֶּלֶת.

2

שָׂמִים כָּל מִילָּה בְּקוּפְסָה מִשֶּׁלָּהּ

בְּקָרוֹב

בִּמְקוֹם רַק לִסְפּוֹר, נִשְׁמוֹר כָּל מִילָּה בְּנִפְרָד — כְּדֵי שֶׁנּוּכַל לְהִשְׁתַּמֵּשׁ בָּהֶן אַחַר כָּךְ.

3

אֵיזוֹ מִילָּה הֲכִי פּוֹפּוּלָרִית בַּסֵּפֶר?

בְּקָרוֹב

סוֹפְרִים כַּמָּה פְּעָמִים כָּל מִילָּה מוֹפִיעָה, וּמַכְרִיזִים עַל מַלְכַּת הַפּוֹפּוּלָרִיּוּת.

4

נוֹתְנִים לַמַּחְשֵׁב סֵפֶר שָׁלֵם לִקְרוֹא

בְּקָרוֹב

עוֹבְרִים מִכַּמָּה מִשְׁפָּטִים קְטַנִּים לְסֵפֶר אֲמִיתִּי, וּמְרִיצִים עָלָיו אֶת כָּל מָה שֶׁבָּנִינוּ.

5

מִי אוֹהֵב לִהְיוֹת לְיַד מִי?

בְּקָרוֹב

מְגַלִּים אֵילוּ שְׁתֵּי מִילִּים אוֹהֲבוֹת לְהוֹפִיעַ אַחַת אַחֲרֵי הַשְּׁנִיָּיה — כְּמוֹ לִמְצוֹא זוּגוֹת חֲבֵרִים טוֹבִים.

6

בּוֹנִים מְנוֹעַ חִיפּוּשׂ מָהִיר לְמִילִּים

בְּקָרוֹב

מְסַדְּרִים אֶת כָּל "הַחֲבֵרוּיוֹת" בֵּין מִילִּים כָּךְ שֶׁאֶפְשָׁר לְחַפֵּשׂ בָּהֶן מַהֵר, גַּם בְּאַלְפֵי מִילִּים.

7

בְּקָרוֹב

03 · פּוֹתְחִים אֶת הַתִּיק

מְשִׂימָה 1 — כַּמָּה מִילִּים יֵשׁ בַּמִּשְׁפָּט?

הַמַּטָּרָה: מַקְלִידִים מִשְׁפָּט, וְהַתּוֹכְנָה עוֹנָה כַּמָּה מִילִּים יֵשׁ בּוֹ. בְּלִי לִסְפּוֹר עַל הָאֶצְבָּעוֹת, וּבְלִי שׁוּם בְּלוֹק קֶסֶם שֶׁכְּבָר יוֹדֵעַ לַעֲשׂוֹת אֶת זֶה — רַק בְּעֶזְרַת עַיִן קְטַנָּה שֶׁעוֹבֶרֶת אוֹת-אַחַר-אוֹת.

הָרַעֲיוֹן לִפְנֵי שֶׁבּוֹנִים

אֵיךְ יוֹדְעִים שֶׁמִּילָּה נִגְמְרָה? פָּשׁוּט — יֵשׁ שֵׁם רֶוַוח! אָז אִם נַעֲבוֹר עַל הַמִּשְׁפָּט אוֹת-אַחַר-אוֹת וְנִסְפּוֹר כַּמָּה רְוָוחִים יֵשׁ, קִיבַּלְנוּ כִּמְעַט אֶת הַתְּשׁוּבָה. חָסֵר רַק דָּבָר אֶחָד קָטָן: הַמִּילָּה הָאַחֲרוֹנָה בַּמִּשְׁפָּט לֹא מִסְתַּיֶּימֶת בְּרֶוַוח — הִיא מִסְתַּיֶּימֶת כִּי הַמִּשְׁפָּט נִגְמַר. אָז צָרִיךְ לְהוֹסִיף עוֹד 1.

שָׁלָב 3 הַסִּיפּוּר שֶׁל הָעַיִן הַקְּטַנָּה

  1. הַתּוֹכְנָה שׁוֹאֶלֶת אֶתְכֶם לִכְתּוֹב מִשְׁפָּט, וְשׁוֹמֶרֶת אוֹתוֹ בַּתֵּיבָה מִשְׁפָּט.
  2. הַתֵּיבָה מִסְפַּר_מִילִּים מַתְחִילָה מ־1 (כִּי גַּם מִשְׁפָּט קְצַרְצַר הוּא לְפָחוֹת מִילָּה אַחַת), וְהָעַיִן שֶׁלָּנוּ מָקוֹם בַּמִּשְׁפָּט מַתְחִילָה עַל הָאוֹת הָרִאשׁוֹנָה.
  3. עַכְשָׁיו מַתְחִיל הַסִּיבּוּב: כָּל עוֹד הָעַיִן עֲדַיִין לֹא עָבְרָה אֶת סוֹף הַמִּשְׁפָּט, הִיא מַמְשִׁיכָה לִבְדּוֹק.
  4. בְּכָל אוֹת, הָעַיִן שׁוֹאֶלֶת: "הַאִם זוֹ רֶוַוח?" אִם כֵּן — זֶה סִימָן שֶׁמִּילָּה הִסְתַּיְּימָה, וּמוֹסִיפִים 1 לְתֵיבַת מִסְפַּר_מִילִּים.
  5. בְּכָל מִקְרֶה — רֶוַוח אוֹ לֹא — הָעַיִן זָזָה צַעַד אֶחָד קָדִימָה, לָאוֹת הַבָּאָה.
  6. כְּשֶׁהָעַיִן מַגִּיעָה לְסוֹף הַמִּשְׁפָּט, הַתֵּיבָה מִסְפַּר_מִילִּים מְכִילָה אֶת הַתְּשׁוּבָה, וְהַתּוֹכְנָה אוֹמֶרֶת אוֹתָהּ.

עַכְשָׁיו אַתֶּם הַבַּלָּשִׁים

אֶתְגָּר קַל: מָה קוֹרֶה אִם יֵשׁ שְׁנֵי רְוָוחִים בְּרֶצֶף בְּאֶמְצַע הַמִּשְׁפָּט? הֵרִיצוּ אֶת הַתּוֹכְנָה וּבִדְקוּ — הִיא תִּסְפּוֹר מִילָּה אַחַת מִדַּי! לָמָּה זֶה קוֹרֶה, וְאֵיךְ הֱיִיתֶם מְתַקְּנִים אֶת זֶה?

אֶתְגָּר לַגִּיבּוֹרִים: הוֹסִיפוּ תֵּיבָה חֲדָשָׁה שֶׁתִּזְכּוֹר מָה הַמִּילָּה הֲכִי אֲרוּכָּה בַּמִּשְׁפָּט. רֶמֶז: תִּצְטָרְכוּ תֵּיבָה שֶׁאוֹסֶפֶת אוֹתִיּוֹת אַחַת-אַחַת (כָּל עוֹד לֹא הִגַּעְתֶּם לְרֶוַוח), וְלִבְדּוֹק אֶת הָאוֹרֶךְ שֶׁלָּהּ בְּכָל פַּעַם שֶׁמַּגִּיעִים לְרֶוַוח חָדָשׁ.
